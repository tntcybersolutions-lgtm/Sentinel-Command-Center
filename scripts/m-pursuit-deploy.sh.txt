path = 'client/src/pages/home-assistant.tsx'
with open(path) as f: src = f.read()

if 'PursuitBadge' in src:
  print("  = home already patched")
else:
  badge_component = '''
// PursuitBadge: morning surface for new SAM opportunities. One row, one click.
function PursuitBadge() {
  const [, navigate] = useLocation();
  const { data } = useQuery<{ data: any[] }>({
    queryKey: ["/api/v4/opportunities", "pursuit-badge"],
    queryFn: async () => {
      const r = await fetch("/api/v4/opportunities?limit=500");
      if (!r.ok) return { data: [] };
      return r.json();
    },
    refetchInterval: 5 * 60 * 1000,
  });
  const opps = data?.data || [];
  const last24h = Date.now() - 24 * 60 * 60 * 1000;
  const fresh = opps.filter((o: any) => !o.decision && (!o.postedAt || new Date(o.postedAt).getTime() >= last24h));
  const count = fresh.length > 0 ? fresh.length : opps.filter((o: any) => !o.decision).length;
  if (count === 0) return null;
  const isFresh = fresh.length > 0;
  return (
    <button onClick={() => navigate("/pursuit")} className="w-full flex items-center gap-3 p-3 rounded-lg border bg-gradient-to-r from-blue-50 to-indigo-50 border-blue-200 hover:from-blue-100 hover:to-indigo-100 transition-all group" data-testid="pursuit-badge">
      <div className="h-9 w-9 rounded-full bg-blue-600 text-white flex items-center justify-center text-lg flex-shrink-0">NEW</div>
      <div className="flex-1 text-left">
        <div className="text-sm font-semibold text-gray-900">
          {count} SAM opportunit{count === 1 ? "y" : "ies"} {isFresh ? "in the last 24h" : "awaiting triage"}
        </div>
        <div className="text-xs text-gray-600">Click to triage — Pursue, Watch, or Skip in one click.</div>
      </div>
      <ChevronRight className="h-5 w-5 text-blue-600 group-hover:translate-x-0.5 transition-transform"/>
    </button>
  );
}

'''
  marker = 'export default function'
  if marker in src:
    src = src.replace(marker, badge_component + marker, 1)
    print("  + PursuitBadge component definition added")

  jsx_marker = '''      {/* Herbie's one-line brief */}
      <HerbieBriefBanner />

      {/* Action Required Today */}'''
  jsx_new = '''      {/* Herbie's one-line brief */}
      <HerbieBriefBanner />

      {/* Pursuit Badge — new SAM opps to triage */}
      <PursuitBadge />

      {/* Action Required Today */}'''
  if jsx_marker in src:
    src = src.replace(jsx_marker, jsx_new, 1)
    print("  + <PursuitBadge/> rendered between brief and action-required")
  else:
    print("  ! HerbieBrief anchor missing — manual review needed")

with open(path,'w') as f: f.write(src)
PYEOF
